المخارج (Exits) — بالأولوية:
  sudden_dump          = +15% ذروة ثم -30% هبوط → خروج كامل
  stop_loss            = -95%   (فعلياً "بدون SL" — دعها تركض حتى الموت)
  mega_tp              = +500%  → خروج كامل
  partial_tp           = +50%   → بيع 50% فقط (تأمين ربح)
  smart_trailing       = +30% ذروة، 25% عرض، يضيق لـ 10% بعد +100%
  take_profit_2        = +200%  → خروج كامل
  stagnation           = 6 ساعات، ذروة < 5% → خروج
  time_exit            = 5 أيام (7200 دقيقة) → خروج نهائي

بوابات الدخول (Entry Gates):
  V511 single-holder   = 30% أقصى (محفظة واحدة تحوز 30% → حجب صارم)
  top-10 concentration = 40% أقصى
  liquidity            = $8000 أدنى
  age                  = 45 ثانية أدنى (تجاوز نافذة rug الأولى)
  pump entry           = chg_5m ∈ [3%, 80%]، age ≤ 10 دقائق
  dip entry            = chg_1h ≤ -10% AND chg_5m ≥ 0 (استقرار بعد تراجع)
  min_buys             = 15
  buy/sell ratio       = 1.2 أدنى

بوابة السوق (Market Regime):
  median_1h > -30% عبر آخر 80 توكن → السماح بالـ pumps
  إذا الـ median ≤ -30% (انهيار سوق): حجب الـ pumps، السماح بالـ dips فقط

الدفاع (Defense):
  daily_loss_stop      = 4% (إيقاف اليوم إذا خسرنا 4% من رأس المال)
  streak_losses_max    = 3 خسائر متتالية → إيقاف يوم كامل
  dd_pause_pct         = 70% من القمة → إيقاف يومين
  dd_half_pct          = 85% من القمة → نصف الحجم
  max_buys_per_day     = 12

التعلّم (Learning):
  بعد كل سلسلة خاسرة (3 خسائر متتالية):
    - تشديد max_chg_5m_pct بـ 20% (أقصى 60% تشديد)
    - هذا يجعل البوت أكثر تحفظاً بعد الخسائر
