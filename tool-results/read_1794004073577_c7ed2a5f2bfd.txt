   104→    def detect_honeypot(self, onchain: Dict, token) -> DangerDetection:
   105→        """Honeypot: you can buy but can't sell."""
   106→        if not onchain or not token:
   107→            return DangerDetection("honeypot", False, "low", 0, "No on-chain data")
   108→
   109→        # Heuristics
   110→        can_sell = onchain.get("can_sell", True)
   111→        sell_tax = onchain.get("sell_tax_estimate_pct", 0)
   112→        freeze_auth = onchain.get("freeze_authority_enabled", False)
   113→
   114→        score = 0
   115→        if not can_sell: score += 50
   116→        if sell_tax > 25: score += 30  # high sell tax
   117→        if freeze_auth: score += 20
   118→
   119→        detected = score >= 50
   120→        severity = "critical" if score >= 80 else ("high" if score >= 50 else "low")
   121→        return DangerDetection(
   122→            "honeypot", detected, severity, score / 100,
   123→            "Honeypot: يمكن الشراء لكن لا يمكن البيع",
   124→            {"can_sell": can_sell, "sell_tax": sell_tax, "freeze_auth": freeze_auth},
   125→            "AVOID" if detected else "SAFE"
   126→        )

   234→    def detect_wash_trading(self, df: pd.DataFrame, token=None) -> DangerDetection:
   235→        """Wash Trading: same wallet buying and selling to fake volume."""
   236→        if df is None or len(df) < 20:
   237→            return DangerDetection("wash_trading", False)
   238→        vols = df["volume"].astype(float).tail(20)
   239→        cv = vols.std() / max(vols.mean(), 1e-9)  # coefficient of variation
   240→        # Suspiciously uniform volume = wash trading
   241→        detected = cv < 0.15
   242→        confidence = max(0, 1 - cv / 0.15) if cv < 0.15 else 0
   243→        return DangerDetection(
   244→            "wash_trading", detected,
   245→            "high" if detected else "low",
   246→            confidence,
   247→            f"حجم تداول منتظم بشكل مريب (CV={cv:.3f})",
   248→            {"cv": cv, "avg_vol": float(vols.mean())},
   249→            "AVOID" if detected else "MONITOR"
   250→        )
   251→
   252→    def detect_fake_volume(self, df: pd.DataFrame, token=None) -> DangerDetection:
   253→        """Fake Volume: high volume but no price movement."""
   254→        if df is None or len(df) < 30 or token is None:
   255→            return DangerDetection("fake_volume", False)
   256→        recent = df.tail(30)
   257→        vol = float(recent["volume"].mean())
   258→        price_range = (recent["high"].max() - recent["low"].min()) / recent["close"].mean()
   259→        # High volume but low price movement = fake
   260→        vol_to_movement_ratio = vol / max(price_range, 1e-9)
   261→        detected = vol_to_movement_ratio > 100000 and price_range < 0.05
   262→        return DangerDetection(
   263→            "fake_volume", detected,
   264→            "medium" if detected else "low",
   265→            0.7 if detected else 0,
   266→            "حجم مرتفع بدون حركة سعر حقيقية",
   267→            {"vol": vol, "price_range": price_range},
   268→            "CAUTION" if detected else "SAFE"
   269→        )

   271→    def detect_spoofing(self, df: pd.DataFrame) -> DangerDetection:
   272→        """Spoofing: large orders placed then cancelled."""
   273→        # Can't detect without order book data, but can infer from volume spikes
   274→        # followed by immediate price reversals
   275→        if df is None or len(df) < 20:
   276→            return DangerDetection("spoofing", False)
   277→        vols = df["volume"].astype(float).tail(20)
   278→        avg_vol = vols.mean()
   279→        spikes = (vols > avg_vol * 3).sum()
   280→        # Many volume spikes without sustained price movement = spoofing
   281→        detected = spikes >= 3
   282→        return DangerDetection(
   283→            "spoofing", detected,
   284→            "medium" if detected else "low",
   285→            min(spikes / 5, 1.0),
   286→            f"{spikes} ارتفاعات حجم مشبوهة في آخر 20 شمعة",
   287→            {"spikes": spikes, "avg_vol": avg_vol},
   288→            "CAUTION" if detected else "SAFE"
   289→        )

   291→    def detect_layering(self, df: pd.DataFrame) -> DangerDetection:

   310→    def detect_front_running(self, df: pd.DataFrame) -> DangerDetection:
   311→        """Front Running: bots buying before large orders."""
   312→        if df is None or len(df) < 30:
   313→            return DangerDetection("front_running", False)
   314→        # Detect: sudden price spike + immediate reversal
   315→        recent = df.tail(10)
   316→        max_gain = (recent["high"].max() - recent["open"].iloc[0]) / recent["open"].iloc[0]
   317→        final_change = (recent["close"].iloc[-1] - recent["open"].iloc[0]) / recent["open"].iloc[0]
   318→        # Spike then reversal = front running
   319→        detected = max_gain > 0.05 and final_change < 0
   320→        return DangerDetection(
   321→            "front_running", detected,
   322→            "medium" if detected else "low",
   323→            0.7 if detected else 0,
   324→            f"ارتفاع {max_gain:.1%} ثم انعكاس",
   325→            {"max_gain": max_gain, "final_change": final_change},
   326→            "CAUTION" if detected else "SAFE"
   327→        )

   351→    def detect_pump_and_dump(self, df: pd.DataFrame) -> DangerDetection:
   352→        """Pump and Dump: rapid rise then crash."""
   353→        if df is None or len(df) < 50:
   354→            return DangerDetection("pump_and_dump", False)
   355→        recent = df.tail(50)
   356→        prices = recent["close"].astype(float)
   357→        max_price = prices.max()
   358→        current_price = prices.iloc[-1]
   359→        max_idx = prices.values.tolist().index(max_price)
   360→        # If peak was in middle and we're now 30%+ below
   361→        drop_from_peak = (max_price - current_price) / max_price
   362→        detected = drop_from_peak > 0.30 and 10 < max_idx < 40
   363→        return DangerDetection(
   364→            "pump_and_dump", detected,
   365→            "high" if detected else "low",
   366→            drop_from_peak if detected else 0,
   367→            f"هبوط {drop_from_peak:.1%} من القمة",
   368→            {"drop_from_peak": drop_from_peak, "peak_position": max_idx},
   369→            "AVOID" if detected else "MONITOR"
   370→        )

   466→    def detect_stop_hunting(self, df: pd.DataFrame) -> DangerDetection:
   467→        """Stop Hunting: price spikes to trigger stops then reverses."""
   468→        if df is None or len(df) < 30:
   469→            return DangerDetection("stop_hunting", False)
   470→        recent = df.tail(30)
   471→        # Detect: wick below recent low, then close back above
   472→        recent_low = recent["low"].iloc[:-1].min()
   473→        last_low = recent["low"].iloc[-1]
   474→        last_close = recent["close"].iloc[-1]
   475→        # Wick below support then close above
   476→        detected = last_low < recent_low * 0.98 and last_close > recent_low
   477→        return DangerDetection(
   478→            "stop_hunting", detected,
   479→            "medium" if detected else "low",
   480→            0.6 if detected else 0,
   481→            "صيد أوامر الإيقاف: اختراق أسفل الدعم ثم إغلاق فوقه",
   482→            {"recent_low": recent_low, "last_low": last_low, "last_close": last_close},
   483→            "CAUTION" if detected else "SAFE"
   484→        )

   486→    def detect_fake_breakout(self, df: pd.DataFrame) -> DangerDetection:
   487→        """Fake Breakout: price breaks resistance then immediately reverses."""
   488→        if df is None or len(df) < 30:
   489→            return DangerDetection("fake_breakout", False)
   490→        recent = df.tail(30)
   491→        resistance = recent["high"].iloc[:-5].max()
   492→        # Check if we broke resistance in last 5 candles but closed below
   493→        last_5 = recent.tail(5)
   494→        broke = (last_5["high"] > resistance).any()
   495→        closed_below = last_5["close"].iloc[-1] < resistance
   496→        detected = broke and closed_below
   497→        return DangerDetection(
   498→            "fake_breakout", detected,
   499→            "high" if detected else "low",
   500→            0.8 if detected else 0,
   501→            "اختراق وهمي: تجاوز المقاومة ثم إغلاق دونها",
   502→            {"resistance": resistance, "broke": broke, "closed_below": closed_below},
   503→            "AVOID" if detected else "CAUTION"
   504→        )

   519→        return DangerDetection(

   521→            severity if detected else "low",
   522→            1 - (liq / 50000) if detected else 0,

   524→            {"liquidity_usd": liq},

   528→    def detect_slippage_trap(self, token) -> DangerDetection:

   530→        if not token:
   531→            return DangerDetection("slippage_trap", False)
   532→        liq = float(token.liquidity_usd or 0) if hasattr(token, 'liquidity_usd') else 0
   533→        vol_24h = float(token.volume_24h_usd or 0) if hasattr(token, 'volume_24h_usd') else 0
   534→        if liq == 0:
   535→            return DangerDetection("slippage_trap", False)
   536→        # Estimate slippage for $1000 trade

   541→            "high" if slippage_pct > 5 else ("medium" if detected else "low"),
   542→            min(slippage_pct / 10, 1.0),
   543→            f"انزلاق سعري متوقع {slippage_pct:.2f}% لصفقة $1000",
   544→            {"slippage_pct": slippage_pct, "liquidity": liq},

   585→    def detect_flash_loan_attack(self, df: pd.DataFrame, onchain: Dict) -> DangerDetection:
   586→        """Flash Loan Attack: borrowed funds used to manipulate price."""
   587→        if df is None or len(df) < 5:
   588→            return DangerDetection("flash_loan_attack", False)
   589→        # Detect: massive volume spike in 1-2 candles with extreme price movement
   590→        recent = df.tail(5)
   591→        vols = recent["volume"].astype(float)
   592→        avg_vol = vols.mean()
   593→        max_vol = vols.max()
   594→        price_change = abs((recent["close"].iloc[-1] - recent["close"].iloc[0]) / recent["close"].iloc[0])
   595→        detected = max_vol > avg_vol * 5 and price_change > 0.20
   596→        return DangerDetection(
   597→            "flash_loan_attack", detected,
   598→            "critical" if detected else "low",
   599→            0.9 if detected else 0,
   600→            f"هجوم Flash Loan محتمل: حجم {max_vol/avg_vol:.1f}x وحركة {price_change:.1%}",
   601→            {"max_vol": max_vol, "avg_vol": avg_vol, "price_change": price_change},
   602→            "AVOID" if detected else "MONITOR"
   603→        )
