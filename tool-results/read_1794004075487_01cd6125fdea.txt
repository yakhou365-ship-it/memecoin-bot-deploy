   104→   104→    def detect_honeypot(self, onchain: Dict, token) -> DangerDetection:
   105→   105→        """Honeypot: you can buy but can't sell."""
   106→   106→        if not onchain or not token:
   107→   107→            return DangerDetection("honeypot", False, "low", 0, "No on-chain data")
   108→   108→
   109→   109→        # Heuristics
   110→   110→        can_sell = onchain.get("can_sell", True)
   111→   111→        sell_tax = onchain.get("sell_tax_estimate_pct", 0)
   112→   112→        freeze_auth = onchain.get("freeze_authority_enabled", False)
   113→   113→
   114→   114→        score = 0
   115→   115→        if not can_sell: score += 50
   116→   116→        if sell_tax > 25: score += 30  # high sell tax
   117→   117→        if freeze_auth: score += 20
   118→   118→
   119→   119→        detected = score >= 50
   120→   120→        severity = "critical" if score >= 80 else ("high" if score >= 50 else "low")
   121→   121→        return DangerDetection(
   122→   122→            "honeypot", detected, severity, score / 100,

   124→   124→            {"can_sell": can_sell, "sell_tax": sell_tax, "freeze_auth": freeze_auth},

   234→   234→    def detect_wash_trading(self, df: pd.DataFrame, token=None) -> DangerDetection:
   235→   235→        """Wash Trading: same wallet buying and selling to fake volume."""
   236→   236→        if df is None or len(df) < 20:
   237→   237→            return DangerDetection("wash_trading", False)
   238→   238→        vols = df["volume"].astype(float).tail(20)
   239→   239→        cv = vols.std() / max(vols.mean(), 1e-9)  # coefficient of variation
   240→   240→        # Suspiciously uniform volume = wash trading
   241→   241→        detected = cv < 0.15
   242→   242→        confidence = max(0, 1 - cv / 0.15) if cv < 0.15 else 0
   243→   243→        return DangerDetection(
   244→   244→            "wash_trading", detected,
   245→   245→            "high" if detected else "low",
   246→   246→            confidence,
   247→   247→            f"حجم تداول منتظم بشكل مريب (CV={cv:.3f})",
   248→   248→            {"cv": cv, "avg_vol": float(vols.mean())},
   249→   249→            "AVOID" if detected else "MONITOR"
   250→   250→        )

   252→   252→    def detect_fake_volume(self, df: pd.DataFrame, token=None) -> DangerDetection:
   253→   253→        """Fake Volume: high volume but no price movement."""
   254→   254→        if df is None or len(df) < 30 or token is None:
   255→   255→            return DangerDetection("fake_volume", False)
   256→   256→        recent = df.tail(30)
   257→   257→        vol = float(recent["volume"].mean())
   258→   258→        price_range = (recent["high"].max() - recent["low"].min()) / recent["close"].mean()
   259→   259→        # High volume but low price movement = fake
   260→   260→        vol_to_movement_ratio = vol / max(price_range, 1e-9)
   261→   261→        detected = vol_to_movement_ratio > 100000 and price_range < 0.05
   262→   262→        return DangerDetection(
   263→   263→            "fake_volume", detected,
   264→   264→            "medium" if detected else "low",
   265→   265→            0.7 if detected else 0,
   266→   266→            "حجم مرتفع بدون حركة سعر حقيقية",
   267→   267→            {"vol": vol, "price_range": price_range},
   268→   268→            "CAUTION" if detected else "SAFE"
   269→   269→        )

   271→   271→    def detect_spoofing(self, df: pd.DataFrame) -> DangerDetection:
   272→   272→        """Spoofing: large orders placed then cancelled."""
   273→   273→        # Can't detect without order book data, but can infer from volume spikes
   274→   274→        # followed by immediate price reversals
   275→   275→        if df is None or len(df) < 20:

   279→   279→        spikes = (vols > avg_vol * 3).sum()

   282→   282→        return DangerDetection(

   284→   284→            "medium" if detected else "low",
   285→   285→            min(spikes / 5, 1.0),
   286→   286→            f"{spikes} ارتفاعات حجم مشبوهة في آخر 20 شمعة",
   287→   287→            {"spikes": spikes, "avg_vol": avg_vol},

   392→   392→    def detect_capitulation(self, df: pd.DataFrame) -> DangerDetection:
   393→   393→        """Capitulation: panic selling with high volume."""
   394→   394→        if df is None or len(df) < 20:
   395→   395→            return DangerDetection("capitulation", False)
   396→   396→        recent = df.tail(20)
   397→   397→        price_change = (recent["close"].iloc[-1] - recent["open"].iloc[0]) / recent["open"].iloc[0]
   398→   398→        vol = recent["volume"].astype(float)
   399→   399→        vol_spike = vol.iloc[-5:].mean() / max(vol.iloc[-20:-5].mean(), 1e-9)
   400→   400→        detected = price_change < -0.20 and vol_spike > 2
   401→   401→        return DangerDetection(
   402→   402→            "capitulation", detected,
