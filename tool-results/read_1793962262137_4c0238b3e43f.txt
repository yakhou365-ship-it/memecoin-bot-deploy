   134→# ============================================================
   135→# Bonding curve math (from pump.fun docs)
   136→# ============================================================
   137→class BondingCurveMath:
   138→    """
   139→    Compute token price from bonding curve reserves.
   140→
   141→    Pump.fun uses a constant-product AMM:
   142→        price = (virtual_sol_reserves * total_supply) /
   143→                (virtual_token_reserves * sol_price_usd)
   144→
   145→    When tokens are bought:
   146→        - real_sol_reserves increases
   147→        - real_token_reserves decreases
   148→        - virtual_sol_reserves stays fixed at 30 SOL
   149→        - virtual_token_reserves stays fixed at 1.073B tokens
   150→
   151→    Graduation happens when real_sol_reserves reaches 85 SOL.
   152→    """
   153→
   154→    @staticmethod
   155→    def compute_price_usd(
   156→        real_sol_reserves: float,
   157→        sol_price_usd: float = SOL_PRICE_USD_FALLBACK,
   158→    ) -> float:
   159→        """
   160→        Compute token price in USD given the current SOL reserves in the curve.
   161→
   162→        Pump.fun uses a constant-product bonding curve:
   163→            k = (virtual_sol + real_sol) * (virtual_tokens - tokens_bought)
   164→
   165→        Initially: k = 30 SOL * 1.073B tokens
   166→        As SOL enters curve: token supply decreases, price increases
   167→        Price per token = (virtual_sol + real_sol) / (virtual_tokens - tokens_bought)
   168→
   169→        At graduation (real_sol=85): price reaches ~$0.00016 (when SOL=$200)
   170→        """
   171→        if real_sol_reserves < 0:
   172→            return 0.0
   173→
   174→        # Compute tokens bought so far from curve
   175→        # When real_sol enters: tokens bought = real_sol / virtual_sol * virtual_tokens (linear approx)
   176→        # But actual pump.fun curve is constant-product, so:
   177→        # k = (virtual_sol + real_sol) * (virtual_tokens - tokens_bought)
   178→        # Initial k = virtual_sol * virtual_tokens
   179→        # So: tokens_bought = virtual_tokens - (virtual_sol * virtual_tokens) / (virtual_sol + real_sol)
   180→        # tokens_bought = virtual_tokens * (1 - virtual_sol / (virtual_sol + real_sol))
   181→        # tokens_bought = virtual_tokens * real_sol / (virtual_sol + real_sol)
   182→
   183→        if real_sol_reserves == 0:
   184→            # Initial price = virtual_sol / virtual_tokens
   185→            price_sol = VIRTUAL_SOL_RESERVES / VIRTUAL_TOKEN_RESERVES
   186→        else:
   187→            # After buying, effective token reserves decrease
   188→            tokens_bought = VIRTUAL_TOKEN_RESERVES * real_sol_reserves / (VIRTUAL_SOL_RESERVES + real_sol_reserves)
   189→            effective_tokens = VIRTUAL_TOKEN_RESERVES - tokens_bought
   190→            effective_sol = VIRTUAL_SOL_RESERVES + real_sol_reserves
   191→
   192→            if effective_tokens <= 0:
   193→                # Curve exhausted (shouldn't happen before graduation)
   194→                return 0.0
   195→
   196→            price_sol = effective_sol / effective_tokens
   197→
   198→        price_usd = price_sol * sol_price_usd
   199→        return price_usd
   200→
   201→    @staticmethod
   202→    def compute_bonding_progress(real_sol_reserves: float) -> float:
   203→        """
   204→        Compute how far the bonding curve has progressed (0-100%).
   205→        Graduation happens at 85 SOL.
   206→        """
   207→        if real_sol_reserves <= 0:
   208→            return 0.0
   209→        progress = (real_sol_reserves / GRADUATION_THRESHOLD_SOL) * 100
   210→        return min(progress, 100.0)
   211→
   212→    @staticmethod
   213→    def compute_market_cap(real_sol_reserves: float, sol_price_usd: float = SOL_PRICE_USD_FALLBACK) -> float:
   214→        """Compute market cap = price * total_supply."""
   215→        price = BondingCurveMath.compute_price_usd(real_sol_reserves, sol_price_usd)
   216→        return price * TOTAL_SUPPLY
   217→
   218→    @staticmethod
   219→    def compute_tokens_for_sol(
   220→        sol_amount: float,
   221→        real_sol_reserves: float,
   222→        fee_bps: int = 100,  # 1% pump.fun fee
   223→    ) -> float:
   224→        """
   225→        How many tokens you get for buying `sol_amount` SOL worth.
   226→
   227→        Uses the constant-product formula:
   228→            new_sol_reserves = real_sol_reserves + sol_amount * (1 - fee/10000)
   229→            tokens_out = effective_tokens_before - effective_tokens_after
   230→
   231→        where: k = effective_sol * effective_tokens = constant
   232→        """
   233→        if sol_amount <= 0 or real_sol_reserves < 0:
   234→            return 0.0
   235→
   236→        # Apply fee
   237→        fee_pct = fee_bps / 10000.0
   238→        sol_after_fee = sol_amount * (1 - fee_pct)
   239→
   240→        # Compute current effective reserves
   241→        if real_sol_reserves == 0:
   242→            effective_sol_before = VIRTUAL_SOL_RESERVES
   243→            effective_tokens_before = VIRTUAL_TOKEN_RESERVES
   244→        else:
   245→            tokens_bought_so_far = VIRTUAL_TOKEN_RESERVES * real_sol_reserves / (VIRTUAL_SOL_RESERVES + real_sol_reserves)
   246→            effective_tokens_before = VIRTUAL_TOKEN_RESERVES - tokens_bought_so_far
   247→            effective_sol_before = VIRTUAL_SOL_RESERVES + real_sol_reserves
   248→
   249→        if effective_tokens_before <= 0:
   250→            return 0.0
   251→
   252→        # Constant k
   253→        k = effective_sol_before * effective_tokens_before
   254→
   255→        # After buy: effective_sol increases, effective_tokens decreases
   256→        effective_sol_after = effective_sol_before + sol_after_fee
   257→        if effective_sol_after <= 0:
   258→            return 0.0
   259→
   260→        effective_tokens_after = k / effective_sol_after
   261→        if effective_tokens_after >= effective_tokens_before:
   262→            return 0.0
   263→
   264→        tokens_out = effective_tokens_before - effective_tokens_after
   265→        return max(tokens_out, 0.0)

   313→    async def _fetch_sol_price(self) -> float:
   314→        """Fetch current SOL price in USD (cached for 60s)."""
   315→        now = time.time()
   316→        if now - self._sol_price_last_fetch < 60:
   317→            return self._sol_price_usd
   318→
   319→        client = await self._get_client()
   320→        try:
   321→            # Jupiter price API
   322→            resp = await client.get(
   323→                "https://lite-api.jup.ag/price/v2",
   324→                params={"ids": "So11111111111111111111111111111111111111112"},
   325→            )
   326→            if resp.status_code == 200:
   327→                data = resp.json()
   328→                price = data.get("data", {}).get(
   329→                    "So11111111111111111111111111111111111111112", {}
   330→                ).get("price")
   331→                if price and float(price) > 0:
   332→                    self._sol_price_usd = float(price)
   333→                    self._sol_price_last_fetch = now
   334→                    return self._sol_price_usd
   335→        except Exception as exc:
   336→            logger.debug(f"SOL price fetch failed: {exc}")
   337→
   338→        return self._sol_price_usd

   466→    # ========================================================
   467→    # Lightning Transaction API (trade execution)
   468→    # ========================================================
   469→    async def _lightning_post(
   470→        self,
   471→        payload: Dict[str, Any],
   472→        timeout: float,
   473→        max_retries: int = 2,
   474→    ):
   475→        """POST to PumpPortal Lightning API with network-level retries."""
   476→        client = await self._get_client()
   477→        url = f"{self.trade_url}?api-key={self.api_key}"
   478→        last_error = ""
   479→        for attempt in range(max_retries + 1):
   480→            try:
   481→                resp = await client.post(url, json=payload, timeout=timeout)
   482→                return resp, None
   483→            except (httpx.TimeoutException, httpx.NetworkError, httpx.ConnectError) as exc:
   484→                last_error = str(exc)[:200]
   485→                logger.warning(
   486→                    f"PumpPortal request attempt {attempt + 1}/{max_retries + 1} failed: {last_error}"
   487→                )
   488→                if attempt < max_retries:
   489→                    await asyncio.sleep(0.5 * (attempt + 1))
   490→        return None, last_error

   492→    async def buy_token(
   493→        self,
   494→        mint_address: str,
   495→        amount_sol: float,
   496→        pool: Optional[str] = None,
   497→    ) -> PumpTradeResult:
   498→        """
   499→        Buy a Pump.fun token with SOL via PumpPortal Lightning API.
   500→
   501→        Args:
   502→            mint_address: Token mint address (Solana)
   503→            amount_sol: SOL amount to spend (e.g., 0.1)
   504→            pool: DEX to trade on (pump, raydium, pump-amm, launchlab, raydium-cpmm, bonk, auto)
   505→
   506→        Returns: PumpTradeResult with tx_signature on success
   507→
   508→        REQUIRES:
   509→        - settings.pumportal_api_key set
   510→        - settings.solana_private_key set (base58 encoded)
   511→        - Wallet has SOL balance
   512→        """
   513→        if not self.api_key:
   514→            return PumpTradeResult(
   515→                success=False,
   516→                error="PumpPortal API key not configured. Set PUMPPORTAL_API_KEY env var.",
   517→                token_address=mint_address,
   518→                side="buy",
   519→                amount_sol=amount_sol,
   520→            )
   521→        if not self.wallet_private_key:
   522→            return PumpTradeResult(
   523→                success=False,
   524→                error="Solana wallet private key not configured. Set SOLANA_PRIVATE_KEY env var.",
   525→                token_address=mint_address,
   526→                side="buy",
   527→                amount_sol=amount_sol,
   528→            )
   529→
   530→        try:
   531→            payload = {

   586→    async def sell_token(
   587→        self,
   588→        mint_address: str,
   589→        amount_tokens: float,
   590→        pool: Optional[str] = None,
   591→    ) -> PumpTradeResult:
   592→        """
   593→        Sell a Pump.fun token for SOL via PumpPortal Lightning API.
   594→
   595→        Args:
   596→            mint_address: Token mint address
   597→            amount_tokens: Number of tokens to sell (or "100%" to sell all)
   598→
   599→        Returns: PumpTradeResult
   600→        """
   601→        if not self.api_key:
   602→            return PumpTradeResult(
   603→                success=False,
   604→                error="PumpPortal API key not configured",
   605→                token_address=mint_address,
   606→                side="sell",
   607→                amount_tokens=amount_tokens,
   608→            )
   609→        if not self.wallet_private_key:
   610→            return PumpTradeResult(
   611→                success=False,
   612→                error="Solana wallet private key not configured",
   613→                token_address=mint_address,
   614→                side="sell",
   615→                amount_tokens=amount_tokens,
   616→            )
   617→
   618→        try:
   619→            payload = {
   620→                "action": "sell",
   621→                "mint": mint_address,
   622→                "amount": amount_tokens,
   623→                "denominatedInSol": "false",
   624→                "slippage": self.slippage_pct,
   625→                "priorityFee": self.priority_fee_sol * 3,
   626→                "skipPreflight": "true",
   627→                "jitoOnly": "true",
   628→                "pool": pool or "pump",
   629→            }
   630→            resp, net_error = await self._lightning_post(payload, timeout=10.0, max_retries=2)
   631→            if net_error:
   632→                return PumpTradeResult(
   633→                    success=False,
   634→                    error=f"Network error after 3 attempts: {net_error}",
   635→                    token_address=mint_address,
   636→                    side="sell",
   637→                    amount_tokens=amount_tokens,
   638→                )
   639→
   640→            # If the sell fails with "insufficient", retry with the full balance ("100%").
   641→            if resp.status_code == 200:
   642→                data = resp.json()
   643→                err_text = str(data.get("errors", data.get("error", ""))).lower()
   644→                if not data.get("signature") and "insufficient" in err_text:
   645→                    logger.info("Sell insufficient balance, retrying with amount='100%'")
   646→                    retry_payload = dict(payload)
   647→                    retry_payload["amount"] = "100%"
   648→                    resp, net_error = await self._lightning_post(retry_payload, timeout=10.0, max_retries=1)
   649→                    if net_error:
   650→                        return PumpTradeResult(
   651→                            success=False,
   652→                            error=f"Retry network error: {net_error}",
   653→                            token_address=mint_address,
   654→                            side="sell",
   655→                            amount_tokens=amount_tokens,
   656→                        )
   657→
   658→            if resp.status_code == 200:
   659→                data = resp.json()
   660→                if data.get("signature"):
   661→                    return PumpTradeResult(
   662→                        success=True,
   663→                        tx_signature=data["signature"],
   664→                        token_address=mint_address,
   665→                        side="sell",
   666→                        amount_tokens=amount_tokens,
   667→                        timestamp=int(time.time()),
   668→                    )
   669→                return PumpTradeResult(
   670→                    success=False,
   671→                    error=data.get("errors", data.get("error", "Unknown error")),
   672→                    token_address=mint_address,
   673→                    side="sell",
   674→                    amount_tokens=amount_tokens,
   675→                )
   676→            return PumpTradeResult(
   677→                success=False,
   678→                error=f"HTTP {resp.status_code}: {resp.text[:200]}",
   679→                token_address=mint_address,
   680→                side="sell",
   681→                amount_tokens=amount_tokens,
   682→            )
   683→        except Exception as exc:
   684→            logger.exception(f"PumpPortal sell failed: {exc}")
   685→            return PumpTradeResult(
   686→                success=False,
   687→                error=str(exc)[:200],
   688→                token_address=mint_address,
   689→                side="sell",
   690→                amount_tokens=amount_tokens,

   693→    async def confirm_signature(
   694→        self,
   695→        signature: str,
   696→        timeout_seconds: float = 30.0,
   697→        max_retries: int = 2,
   698→    ) -> Dict[str, Any]:
   699→        """Confirm a Solana transaction signature via RPC.
   700→
   701→        Fast polling (0.5s), short RPC timeout (3s), accepts any of
   702→        processed/confirmed/finalized as success.
   703→        """
   704→        if not signature:
   705→            return {"confirmed": False, "error": "empty signature"}
   706→
   707→        client = await self._get_client()
   708→        deadline = time.time() + timeout_seconds
   709→        accepted = ("processed", "confirmed", "finalized")
   710→        last_status = "unknown"
   711→        rpc_url = settings.solana_rpc_url
   712→
   713→        while time.time() < deadline:
   714→            try:
   715→                resp = await client.post(
   716→                    rpc_url,
   717→                    json={
   718→                        "jsonrpc": "2.0",
   719→                        "id": 1,
   720→                        "method": "getSignatureStatuses",
   721→                        "params": [[signature], {"searchTransactionHistory": True}],
   722→                    },
   723→                    timeout=3.0,
   724→                )
   725→                if resp.status_code == 200:
   726→                    data = resp.json()
   727→                    value = (data.get("result", {}) or {}).get("value", [])
   728→                    info = value[0] if value else None
   729→                    if info:
   730→                        # V407 P1-10: Check err FIRST — a tx can be "processed" but failed
   731→                        if info.get("err"):
   732→                            return {
   733→                                "confirmed": False,
   734→                                "error": str(info.get("err")),
   735→                                "status": info.get("confirmationStatus", "unknown"),
   736→                                "signature": signature,
   737→                            }
   738→                        last_status = info.get("confirmationStatus", "unknown")
   739→                        if last_status in accepted:
   740→                            return {
   741→                                "confirmed": True,
   742→                                "status": last_status,
   743→                                "slot": info.get("slot"),
   744→                                "err": None,
   745→                                "signature": signature,
   746→                            }
   747→                        if info.get("err"):
   748→                            return {
   749→                                "confirmed": False,
   750→                                "error": str(info.get("err")),
   751→                                "status": last_status,
   752→                                "signature": signature,
   753→                            }
   754→            except Exception as exc:
   755→                logger.debug(f"getSignatureStatuses attempt failed: {str(exc)[:150]}")
   756→
   757→            await asyncio.sleep(0.5)
   758→
   759→        return {
   760→            "confirmed": False,
   761→            "error": f"timeout after {timeout_seconds}s (last status: {last_status})",
   762→            "status": last_status,
   763→            "signature": signature,
   764→        }

   769→    async def get_wallet_balance(self) -> Dict[str, float]:
   770→        """Get wallet SOL and token balances."""
   771→        if not self.wallet_address:
   772→            return {"sol": 0.0, "tokens": []}
   773→
   774→        client = await self._get_client()
   775→        try:
   776→            # Use Solana RPC to get balance
   777→            resp = await client.post(
   778→                settings.solana_rpc_url,
   779→                json={
   780→                    "jsonrpc": "2.0",
   781→                    "id": 1,
   782→                    "method": "getBalance",
   783→                    "params": [self.wallet_address],
   784→                },
   785→            )
   786→            if resp.status_code == 200:
   787→                data = resp.json()
   788→                lamports = data.get("result", {}).get("value", 0)
   789→                sol_balance = lamports / 1_000_000_000  # 1 SOL = 1e9 lamports
   790→                return {"sol": sol_balance, "tokens": []}
   791→        except Exception as exc:
   792→            logger.debug(f"Wallet balance failed: {exc}")
   793→        return {"sol": 0.0, "tokens": []}

   835→    def _parse_token(self, data: Dict[str, Any]) -> Optional[PumpFunToken]:
   836→        """Parse a token dict from PumpPortal API into PumpFunToken."""
   837→        if not data or not isinstance(data, dict):
   838→            return None
   839→
   840→        try:
   841→            sol_price = self._sol_price_usd
   842→            real_sol = float(data.get("sol_reserves", 0) or 0)
   843→            real_tokens = float(data.get("token_reserves", 0) or 0)
   844→
   845→            # Compute derived fields if not provided
   846→            market_cap = float(data.get("market_cap", 0) or 0)
   847→            price_usd = float(data.get("usd_market_cap", 0) or 0) / TOTAL_SUPPLY if not market_cap else market_cap / TOTAL_SUPPLY
   848→
   849→            # Use bonding curve math if market cap is 0
   850→            if market_cap <= 0 and real_sol > 0:
   851→                market_cap = BondingCurveMath.compute_market_cap(real_sol, sol_price)
   852→                price_usd = market_cap / TOTAL_SUPPLY
   853→
   854→            progress = float(data.get("bonding_curve_progress", 0) or 0)
   855→            if progress <= 0 and real_sol > 0:
   856→                progress = BondingCurveMath.compute_bonding_progress(real_sol)
   857→
   858→            return PumpFunToken(
   859→                address=data.get("mint", "") or data.get("address", ""),
   860→                symbol=data.get("symbol", ""),
   861→                name=data.get("name", ""),
   862→                description=data.get("description", ""),
   863→                image_uri=data.get("image_uri", ""),
   864→                creator=data.get("creator", ""),
   865→                created_timestamp=int(data.get("created_timestamp", 0) or 0),
   866→                market_cap_usd=market_cap,
   867→                price_usd=price_usd,
   868→                price_sol=price_usd / sol_price if sol_price > 0 else 0,
   869→                bonding_curve_progress=progress,
   870→                virtual_sol_reserves=float(data.get("virtual_sol_reserves", VIRTUAL_SOL_RESERVES) or VIRTUAL_SOL_RESERVES),
   871→                virtual_token_reserves=float(data.get("virtual_token_reserves", VIRTUAL_TOKEN_RESERVES) or VIRTUAL_TOKEN_RESERVES),
   872→                real_sol_reserves=real_sol,
   873→                real_token_reserves=real_tokens,
   874→                total_supply=float(data.get("total_supply", TOTAL_SUPPLY) or TOTAL_SUPPLY),
   875→                website=data.get("website", ""),
   876→                twitter=data.get("twitter", ""),
   877→                telegram=data.get("telegram", ""),
   878→                is_graduated=bool(data.get("complete", False) or data.get("is_graduated", False)),
   879→                raydium_pool=data.get("raydium_pool", ""),
   880→                holder_count=int(data.get("holder_count", 0) or 0),
   881→                volume_24h_usd=float(data.get("volume_24h", 0) or 0),
   882→                volume_5m_usd=float(data.get("volume_5m", 0) or 0),
   883→                price_change_24h_pct=float(data.get("price_change_24h", 0) or 0),
   884→            )
   885→        except Exception as exc:
   886→            logger.debug(f"PumpFun token parse failed: {exc}")
   887→            return None
